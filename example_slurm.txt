#!/bin/bash
# ==============================================================================
# APPENDIX: Automated Slurm Job Generator and Queue Submission Wrapper
# ==============================================================================
# Description: Generates a generalized Slurm execution script for PHITS Monte Carlo
#              sweeps and submits it to the HPC queue scheduler.
# ==============================================================================

# 1. Define Execution Parameters
PARTICLE="proton"
HISTORIES=50000000
BATCHES=16
CPUS=8
WALLTIME="36:00:00"

# Target Incident Energy Array Definition (MeV)
# Update array values to scale the energy sweep across the desired range
ENERGIES=(10 15 20 25 30 40 50 60 80 100)

# Create local directories for Slurm stdout/stderr logs
mkdir -p ./logs

# 2. Dynamically Write the Slurm Submission File
cat > submit_sweep.slurm << EOF
#!/bin/bash --login
#SBATCH -p multicore                                       # Target cluster queue partition
#SBATCH --ntasks=1                                         # Task allocation count
#SBATCH --cpus-per-task=${CPUS}                             # OpenMP CPU core allocation
#SBATCH -t ${WALLTIME}                                     # Maximum execution walltime (hh:mm:ss)
#SBATCH -o ./logs/phits.o%A_%a                             # Relative standard output log path
#SBATCH -e ./logs/phits.e%A_%a                             # Relative standard error log path
#SBATCH --array=0-$((${#ENERGIES[@]} - 1))                  # Dynamically sets array bounds (0 to N-1)

# Environment Module Loading
module load compilers/intel/oneapi/2025.3.1
module load compiler

# Export OpenMP Thread Count based on allocated Slurm CPUs
export OMP_NUM_THREADS=\$SLURM_CPUS_PER_TASK

# Extract current array task energy value
ENERGIES=(${ENERGIES[*]})
E=\${ENERGIES[\$SLURM_ARRAY_TASK_ID]}

# Execute PHITS Python automation pipeline wrapper
python3 run_energy.py ${PARTICLE} \$E ${HISTORIES} ${BATCHES}
EOF

# 3. Submit generated job script to the Slurm scheduler
sbatch submit_sweep.slurm
